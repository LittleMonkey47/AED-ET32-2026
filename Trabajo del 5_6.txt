ejercicio 12:


using System;


class Program
{
    static void Main()
    {
        string[] nombres = new string[100];


        int cantidad = 0;
        bool repetido = false;


        while (repetido == false)
        {
            Console.Write("Ingrese un nombre: ");
            string nombre = Console.ReadLine();


            for (int i = 0; i < cantidad; i++)
            {
                if (nombres[i] == nombre)
                {
                    repetido = true;
                }
            }


            if (repetido == false)
            {
                nombres[cantidad] = nombre;
                cantidad++;
            }
        }


        Console.WriteLine("Se ingresaron " + cantidad + " nombres antes del repetido.");
    }
}

using System;


class Program
{
    static void Main()
    {
        int jugador1 = 0;
        int jugador2 = 0;


        for (int ronda = 1; ronda <= 3; ronda++)
        {
            Console.Write("Jugador 1 (piedra/papel/tijera): ");
            string j1 = Console.ReadLine();


            Console.Write("Jugador 2 (piedra/papel/tijera): ");
            string j2 = Console.ReadLine();


            if (
                (j1 == "piedra" && j2 == "tijera") ||
                (j1 == "papel" && j2 == "piedra") ||
                (j1 == "tijera" && j2 == "papel")
               )
            {
                jugador1++;
            }
            else if (
                (j2 == "piedra" && j1 == "tijera") ||
                (j2 == "papel" && j1 == "piedra") ||
                (j2 == "tijera" && j1 == "papel")
                    )
            {
                jugador2++;
            }
        }


        Console.WriteLine("Jugador 1 ganó: " + jugador1);
        Console.WriteLine("Jugador 2 ganó: " + jugador2);
    }
}




ejercicio 18:


using System;


class Program
{
    static void Main()
    {
        string mejorNombre = "";
        double mejorNota = 0;
        double suma = 0;


        for (int i = 1; i <= 5; i++)
        {
            Console.Write("Nombre: ");
            string nombre = Console.ReadLine();


            Console.Write("Nota: ");
            double nota = Convert.ToDouble(Console.ReadLine());


            suma = suma + nota;


            if (i == 1 || nota > mejorNota)
            {
                mejorNota = nota;
                mejorNombre = nombre;
            }
        }


        Console.WriteLine("Promedio: " + (suma / 5));
        Console.WriteLine("Mejor alumno: " + mejorNombre);
        Console.WriteLine("Mejor nota: " + mejorNota);
    }
}
